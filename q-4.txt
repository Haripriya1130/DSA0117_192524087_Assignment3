
#include <iostream>
#include <iomanip>
#include <cstdlib>
using namespace std;

class Money {
    long long paise;

public:
    Money(long long rupees = 0, int p = 0)
        : paise(rupees * 100 + p) {}

    Money operator+(const Money& m) const {
        Money temp;
        temp.paise = paise + m.paise;
        return temp;
    }

    Money operator-(const Money& m) const {
        Money temp;
        temp.paise = paise - m.paise;
        return temp;
    }

    Money& operator+=(const Money& m) {
        paise += m.paise;
        return *this;
    }

    bool operator==(const Money& m) const {
        return paise == m.paise;
    }

    bool operator<(const Money& m) const {
        return paise < m.paise;
    }

    friend ostream& operator<<(ostream& out, const Money& m) {
        long long amount = llabs(m.paise);

        if (m.paise < 0)
            out << "-";

        out << "Rs " << amount / 100 << "."
            << setw(2) << setfill('0') << amount % 100;

        return out;
    }

    friend istream& operator>>(istream& in, Money& m) {
        long long r;
        int p;

        cout << "Enter rupees and paise: ";
        in >> r >> p;

        m.paise = r * 100 + p;
        return in;
    }
};

int main() {
    Money a(120, 75);
    Money b(79, 25);

    Money c = a + b;

    cout << "Addition: " << c << endl;

    Money d(0, 75);
    Money e(0, 50);

    cout << "Paise Carry: " << d + e << endl;

    cout << "Subtraction: " << a - b << endl;

    a += b;
    cout << "After += : " << a << endl;

    cout << boolalpha;

    cout << "Less than: " << (b < a) << endl;

    Money input;
    cin >> input;

    cout << "Entered Amount: " << input << endl;

    return 0;
}
