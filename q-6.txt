
#include <iostream>
#include <cstring>
using namespace std;

class Text {
    char* data;

public:
    // Constructor
    Text(const char* str = "") {
        data = new char[strlen(str) + 1];
        strcpy(data, str);
    }

    // Copy Constructor
    Text(const Text& t) {
        data = new char[strlen(t.data) + 1];
        strcpy(data, t.data);
    }

    // Copy Assignment
    Text& operator=(const Text& t) {
        if (this != &t) {
            char* temp = new char[strlen(t.data) + 1];
            strcpy(temp, t.data);

            delete[] data;
            data = temp;
        }

        return *this;
    }

    // Concatenation
    Text operator+(const Text& t) const {
        int len = strlen(data) + strlen(t.data);

        char* temp = new char[len + 1];

        strcpy(temp, data);
        strcat(temp, t.data);

        Text result(temp);

        delete[] temp;

        return result;
    }

    // Equality
    bool operator==(const Text& t) const {
        return strcmp(data, t.data) == 0;
    }

    // Inequality
    bool operator!=(const Text& t) const {
        return !(*this == t);
    }

    // Display
    friend ostream& operator<<(ostream& out, const Text& t) {
        out << t.data;
        return out;
    }

    // Destructor
    ~Text() {
        delete[] data;
    }
};

int main() {
    Text a("Hello ");
    Text b("World");
    Text c("!");

    Text result = a + b + c;

    cout << "First Text: " << a << endl;
    cout << "Second Text: " << b << endl;
    cout << "Combined Text: " << result << endl;

    a = a;

    cout << "After Self Assignment: " << a << endl;

    Text empty("");
    cout << "Empty Text: [" << empty << "]" << endl;

    return 0;
}
